# Iris I2 — Shared Behaviour Fixtures Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every pure domain module's behaviour is captured as language-neutral JSON test vectors in `shared/fixtures/`, and the SQLite schema as `shared/schema/schema.sql`. Jest proves the TS app still matches them, and I3's XCTest runner will prove the Swift port does.

**Architecture:** Cases are authored in TS (`shared/fixtures/cases/<module>.ts`: name, function, arguments). One jest file runs every case through a registry of the real `src/domain` exports, under `TZ=UTC`. In record mode (`npm run fixtures:record`) it writes the results to `shared/fixtures/<module>.json`. In normal mode it fails if the committed JSON differs from what TS now produces. Swift reads only the JSON. The schema snapshot works the same way, using the existing in-memory SQLite test driver.

**Tech Stack:** TypeScript (strict), Jest (jest-expo), better-sqlite3 via `test/memoryDriver.ts`.

**Spec:** `docs/superpowers/specs/2026-10-06-iris-design.md` §3 (shared fixtures, schema fixture, new-feature rule), §8 I2.

## Global Constraints

- One fixture file per pure domain module: `advance`, `shelf`, `rating`, `seasons`, `formatters`, `validate`, `seriesTitle`, `genres`, `mode`, `whatsNew` (spec §3).
- "Fixtures are *extracted* from [the existing tests] (and added to), not a replacement": the existing `src/domain/__tests__` stay untouched.
- "Dates are passed as ISO strings and a fixed `now`, so no fixture depends on the clock." Every case runs with `TZ=UTC`, and the JSON says so.

## The JSON contract (what I3's Swift runner reads)

```jsonc
// shared/fixtures/whatsNew.json
{
  "module": "whatsNew",
  "generatedBy": "npm run fixtures:record",
  "timezone": "UTC",
  "cases": [
    { "name": "…", "fn": "announcementFor", "args": [ … ], "expect": … },
    { "name": "…", "fn": "assertOrdinal",   "args": [ … ], "throws": "exact error message" }
  ]
}
```

- `args` is the positional argument list, exactly as passed to the TS function.
- `expect` is the return value. `null` means the function returned `null` or `undefined`. A `Map` is serialised as a JSON object in insertion order. An object key whose value is `undefined` is absent; Swift treats an absent key as `nil`.
- `throws` replaces `expect` when the function threw. It holds the error's exact `message`.
- Numbers: Swift compares with an absolute tolerance of `1e-9`, and JS's own values are recorded exactly.
- A composite function that isn't a TS export, `rankingScenario`, is defined in the registry (Task 3). Swift implements the same composite in its runner.

## Review Focus

1. **The recording machine's timezone** (the user is in America/Denver, CI would be UTC): the recorded JSON must be identical, and the TZ change must not leak into other test files sharing the jest worker. Pinned in Task 1, Step 1.
2. **TS behaviour changes and nobody re-records**: the suite fails and names the module and case. Pinned in Task 1, Step 6.
3. **A case whose function throws** (`validate`'s asserts): it is recorded as `throws` data and must not crash the runner. Pinned in Task 2, Step 1.
4. **Non-JSON return values** (`scoresFor` returns a `Map`; optional fields may be `undefined`): they must serialise deterministically, not as `{}`. Pinned in Task 3, Step 1.
5. **Coverage drift**: a new test in `src/domain/__tests__` with no matching fixture case. The coverage check fails. Pinned in Task 1, Step 1.

**How to author a module's cases (Tasks 2–3).** Open its test file(s). For every `test(...)`, write at least one case with the same inputs and a `name` that starts with the test's title. When one test makes several `expect` calls on different inputs, write one case per call, suffixed ` (2)`, ` (3)`. Copy the test's local builders (e.g. a `makeEntry` helper) into the cases file as plain functions, so the args are concrete objects. **Don't write expected values by hand**: record mode computes them from TS. Then read the recorded JSON and check that each `expect` agrees with the original test's assertion. A disagreement means the case's inputs are wrong.

```ts
import * as fs from 'fs';
import * as path from 'path';
import { normalize, registry } from '../registry';
import type { FixtureCase } from '../types';
import { cases as whatsNew } from '../cases/whatsNew';

/** Module → its authored cases and the src/domain tests they extract from. */
const MODULES: Record<string, { cases: FixtureCase[]; testFiles: string[] }> = {
  whatsNew: { cases: whatsNew, testFiles: ['whatsNew.test.ts'] },
};

const FIXTURES = path.resolve(__dirname, '..');
const DOMAIN_TESTS = path.resolve(__dirname, '../../../src/domain/__tests__');
const RECORD = process.env.RECORD_FIXTURES === '1';

// Spec §3: fixtures never depend on the machine's clock or timezone. Node
// honours a runtime TZ change; restore it so other files in this jest
// worker are unaffected.
const originalTz = process.env.TZ;
beforeAll(() => {
  process.env.TZ = 'UTC';
});
afterAll(() => {
  if (originalTz === undefined) delete process.env.TZ;
  else process.env.TZ = originalTz;
});

function run(c: FixtureCase): Record<string, unknown> {
  const fn = registry[c.fn];
  if (!fn) throw new Error(`No registry entry for "${c.fn}" (case "${c.name}")`);
  try {
    return { name: c.name, fn: c.fn, args: c.args, expect: normalize(fn(...structuredClone(c.args))) };
  } catch (e) {
    return { name: c.name, fn: c.fn, args: c.args, throws: (e as Error).message };
  }
}

test('cases run under UTC', () => {
  expect(new Date('2026-01-01T00:30:00Z').getDate()).toBe(1);
  expect(new Date('2026-01-01T00:30:00Z').getTimezoneOffset()).toBe(0);
});

describe.each(Object.entries(MODULES))('%s', (module, { cases, testFiles }) => {
  test('case names are unique', () => {
    const names = cases.map((c) => c.name);
    expect(names.filter((n, i) => names.indexOf(n) !== i)).toEqual([]);
  });

  test('covers every test in its src/domain test files', () => {
    const floor = testFiles
      .map((f) => fs.readFileSync(path.join(DOMAIN_TESTS, f), 'utf8').match(/^\s*(it|test)\(/gm)?.length ?? 0)
      .reduce((a, b) => a + b, 0);
    expect(cases.length).toBeGreaterThanOrEqual(floor);
  });

  test(RECORD ? 'records shared/fixtures JSON' : 'matches the committed JSON (re-run `npm run fixtures:record` if TS changed on purpose)', () => {
    const file = path.join(FIXTURES, `${module}.json`);
    const doc = { module, generatedBy: 'npm run fixtures:record', timezone: 'UTC', cases: cases.map(run) };
    if (RECORD) {
      fs.writeFileSync(file, `${JSON.stringify(doc, null, 2)}\n`);
      return;
    }
    expect(fs.existsSync(file)).toBe(true);
    const committed = JSON.parse(fs.readFileSync(file, 'utf8')) as typeof doc;
    // Per case, so a failure names the case that changed.
    expect(committed.cases.map((c) => c.name)).toEqual(doc.cases.map((c) => c.name));
    doc.cases.forEach((actual, i) => expect({ case: actual.name, ...committed.cases[i] }).toEqual({ case: actual.name, ...JSON.parse(JSON.stringify(actual)) }));
  });
});
```

```ts
import { announcementFor } from '@/domain/whatsNew';

/* eslint-disable @typescript-eslint/no-explicit-any */
type AnyFn = (...args: any[]) => unknown;

/** Every function a fixture may name. Swift's runner (I3) mirrors this table. */
export const registry: Record<string, AnyFn> = {
  announcementFor,
};

/** JSON-safe, deterministic: Maps become objects (insertion order), undefined becomes null at the top level. */
export function normalize(value: unknown): unknown {
  if (value === undefined) return null;
  if (value instanceof Map) return Object.fromEntries([...value.entries()].map(([k, v]) => [String(k), normalize(v)]));
  if (Array.isArray(value)) return value.map(normalize);
  if (value !== null && typeof value === 'object') {
    return Object.fromEntries(Object.entries(value).filter(([, v]) => v !== undefined).map(([k, v]) => [k, normalize(v)]));
  }
  return value;
}
```

import { answer, isPlaced, startRanking, type Answer, type RankedItem, type RatingProfile, type Sentiment } from '@/domain/rating';

/** A whole ranking session as one vector: the opponents asked, in order,
 * and where the candidate lands. Not a TS export; Swift mirrors it (I3). */
function rankingScenario(candidate: RatingProfile, sentiment: Sentiment, ranking: RankedItem[], answers: Answer[]) {
  let session = startRanking(candidate, sentiment, ranking);
  const opponents: string[] = [];
  for (const choice of answers) {
    if (session.opponent === null) break;
    opponents.push(session.bucket[session.opponent]!.key);
    session = answer(session, choice);
  }
  return { opponents, lo: session.lo, comparisons: session.comparisons, placed: isPlaced(session) };
}

```ts
import * as fs from 'fs';
import * as path from 'path';
import { migrate } from '@/db/schema';
import { createMemoryDriver } from '../../../test/memoryDriver';

const FILE = path.resolve(__dirname, '../schema.sql');

async function dump(): Promise<string> {
  const db = createMemoryDriver();
  await migrate(db);
  const [{ version }] = (await db.all<{ version: number }>('SELECT version FROM schema_version')) as [{ version: number }];
  const rows = await db.all<{ sql: string }>(
    "SELECT sql FROM sqlite_master WHERE sql IS NOT NULL AND name NOT LIKE 'sqlite_%' ORDER BY type, name",
  );
  return [
    '-- GENERATED by shared/schema/__tests__/schema.test.ts (npm run fixtures:record) — do not edit.',
    '-- The schema after every migration in src/db/schema.ts. Iris I4 must reproduce it.',
    `-- schema_version: ${version}`,
    '',
    rows.map((r) => `${r.sql.trim()};`).join('\n\n'),
    '',
  ].join('\n');
}

test(process.env.RECORD_FIXTURES === '1' ? 'records shared/schema/schema.sql' : 'matches shared/schema/schema.sql (re-run `npm run fixtures:record` after a migration)', async () => {
  const actual = await dump();
  if (process.env.RECORD_FIXTURES === '1') {
    fs.writeFileSync(FILE, actual);
    return;
  }
  expect(fs.existsSync(FILE)).toBe(true);
  expect(fs.readFileSync(FILE, 'utf8')).toBe(actual);
});

test('the snapshot is at the schema version the Swift side targets', async () => {
  expect(await dump()).toContain('-- schema_version: 10');
});
```
